Jeg fikk en feil på 'unrecognized arguments'. Erstatt HELE codemagic.yaml med denne koden. Jeg har rettet feilen i Step 8 og beholdt den fungerende signeringen fra Step 7. Dette er den korrekte syntaksen:

YAML
workflows:
  ios-release:
    name: iOS Release Build
    instance_type: mac_mini_m2
    integrations:
      app_store_connect: Codemagic
    environment:
      node: 20
      vars:
        XCODE_WORKSPACE: "ios/App/App.xcworkspace"
        XCODE_SCHEME: "App"
        BUNDLE_ID: "no.nestwork.vaktapp"
    scripts:
      - name: Install npm dependencies
        script: npm install
      - name: Build web assets
        script: npm run build
      - name: Sync Capacitor
        script: |
          rm -rf ios
          npx cap add ios
          find ios -type f -name "Podfile" -exec sed -i '' "s/platform :ios, '[0-9.]*'/platform :ios, '15.0'/g" {} +
          find ios -type f -name "*.pbxproj" -exec sed -i '' 's/IPHONEOS_DEPLOYMENT_TARGET = [0-9.]*/IPHONEOS_DEPLOYMENT_TARGET = 15.0/g' {} +
          npx cap sync ios
      - name: Install CocoaPods
        script: |
          cd ios/App
          pod install
      - name: Keychain initialize
        script: keychain initialize
      - name: Force Signing Setup
        script: |
          # Denne delen fungerer (Sertifikat + Profil) - ikke rør
          openssl genrsa -out ios_dist.key 2048
          keychain add-certificates --key ios_dist.key
          app-store-connect fetch-signing-files "$BUNDLE_ID" --type IOS_APP_STORE --create --delete-stale-profiles
          keychain add-certificates
      - name: Link Profiles to Xcode
        script: |
          # Her var feilen: --bundle-id skal brukes her for å koble profilen til prosjektet
          xcode-project use-profiles --bundle-id "$BUNDLE_ID"
      - name: Build iOS IPA
        script: |
          # Denne kommandoen er nå renset for ugyldige flagg
          xcode-project build-ipa \
            --workspace "$XCODE_WORKSPACE" \
            --scheme "$XCODE_SCHEME"
    artifacts:
      - build/ios/ipa/*.ipa
    publishing:
      app_store_connect:
        auth: integration
        submit_to_testflight: true
