Vi har nå en helt ren Apple-konto. Erstatt HELE codemagic.yaml med denne koden. Vi bruker --save-flagget som lager både sertifikat og privat nøkkel automatisk, og vi beholder use-profiles for å unngå Error 65:

YAML
workflows:
  ios-release:
    name: iOS Release Build
    instance_type: mac_mini_m2
    integrations:
      app_store_connect: Codemagic
    environment:
      node: 20
      vars:
        XCODE_WORKSPACE: "ios/App/App.xcworkspace"
        XCODE_SCHEME: "App"
        BUNDLE_ID: "no.nestwork.vaktapp"
    scripts:
      - name: Install npm dependencies
        script: npm install
      - name: Build web assets
        script: npm run build
      - name: Sync Capacitor
        script: |
          rm -rf ios
          npx cap add ios
          find ios -type f -name "Podfile" -exec sed -i '' "s/platform :ios, '[0-9.]*'/platform :ios, '15.0'/g" {} +
          find ios -type f -name "*.pbxproj" -exec sed -i '' 's/IPHONEOS_DEPLOYMENT_TARGET = [0-9.]*/IPHONEOS_DEPLOYMENT_TARGET = 15.0/g' {} +
          npx cap sync ios
      - name: Install CocoaPods
        script: |
          cd ios/App
          pod install
      - name: Keychain initialize
        script: keychain initialize
      - name: Fetch and Create Signing Files
        script: |
          # 1. Lag sertifikat og privat nøkkel automatisk og lagre i keychain
          app-store-connect certificates create --type IOS_DISTRIBUTION --save
          
          # 2. Hent eller lag profil (den vil nå finne den nye nøkkelen automatisk)
          app-store-connect fetch-signing-files "$BUNDLE_ID" --type IOS_APP_STORE --create
          
          # 3. Legg alt til i nøkkelringen
          keychain add-certificates
      - name: Link Profiles to Xcode
        script: |
          # Dette fikser Error 65 (provisioning profile error)
          xcode-project use-profiles
      - name: Build iOS IPA
        script: |
          xcode-project build-ipa \
            --workspace "$XCODE_WORKSPACE" \
            --scheme "$XCODE_SCHEME"
    artifacts:
      - build/ios/ipa/*.ipa
    publishing:
      app_store_connect:
        auth: integration
        submit_to_testflight: true
