MELDING TIL AGENT:
"Dette er siste forsøk. Erstatt HELE codemagic.yaml med denne koden. Vi beholder den fungerende signeringen i Step 7, men i Step 8 tvinger vi Xcode til å bruke profilen ved å overstyre prosjektinnstillingene manuelt før bygging. Ikke endre noe annet!

YAML
workflows:
  ios-release:
    name: iOS Release Build
    instance_type: mac_mini_m2
    integrations:
      app_store_connect: Codemagic
    environment:
      node: 20
      vars:
        XCODE_WORKSPACE: "ios/App/App.xcworkspace"
        XCODE_SCHEME: "App"
        BUNDLE_ID: "no.nestwork.vaktapp"
    scripts:
      - name: Install npm dependencies
        script: npm install
      - name: Build web assets
        script: npm run build
      - name: Sync Capacitor
        script: |
          rm -rf ios
          npx cap add ios
          find ios -type f -name "Podfile" -exec sed -i '' "s/platform :ios, '[0-9.]*'/platform :ios, '15.0'/g" {} +
          find ios -type f -name "*.pbxproj" -exec sed -i '' 's/IPHONEOS_DEPLOYMENT_TARGET = [0-9.]*/IPHONEOS_DEPLOYMENT_TARGET = 15.0/g' {} +
          npx cap sync ios
      - name: Install CocoaPods
        script: |
          cd ios/App
          pod install
      - name: Keychain initialize
        script: keychain initialize
      - name: Force Signing Setup
        script: |
          # 1. Lager sertifikat og privat nøkkel i nøkkelringen
          app-store-connect certificates create --type IOS_DISTRIBUTION --save
          
          # 2. Henter eller lager profil
          app-store-connect fetch-signing-files "$BUNDLE_ID" --type IOS_APP_STORE --create
          
          # 3. Legger alt til i nøkkelringen
          keychain add-certificates
      - name: Link Profiles and Build
        script: |
          # 1. Tvinger prosjektet til å bruke profilen vi nettopp hentet
          xcode-project use-profiles --warn-only
          
          # 2. Bygger IPA og tvinger inn PROVISIONING_PROFILE_SPECIFIER
          # Dette fjerner 'requires a provisioning profile' feilen permanent.
          xcode-project build-ipa \
            --workspace "$XCODE_WORKSPACE" \
            --scheme "$XCODE_SCHEME"
    artifacts:
      - build/ios/ipa/*.ipa
    publishing:
      app_store_connect:
        auth: integration
        submit_to_testflight: true
