"Dette er siste sjanse. Erstatt HELE codemagic.yaml med denne koden. Vi bruker de offisielle kommandoene for å generere sertifikat og profil i ett steg. Ikke endre eller legg til noe annet:

YAML
workflows:
  ios-release:
    name: iOS Release Build
    instance_type: mac_mini_m2
    integrations:
      app_store_connect: Codemagic
    environment:
      node: 20
      vars:
        XCODE_WORKSPACE: "ios/App/App.xcworkspace"
        XCODE_SCHEME: "App"
        BUNDLE_ID: "no.nestwork.vaktapp"
    scripts:
      - name: Install npm dependencies
        script: npm install
      - name: Build web assets
        script: npm run build
      - name: Sync Capacitor
        script: |
          rm -rf ios
          npx cap add ios
          find ios -type f -name "Podfile" -exec sed -i '' "s/platform :ios, '[0-9.]*'/platform :ios, '15.0'/g" {} +
          find ios -type f -name "*.pbxproj" -exec sed -i '' 's/IPHONEOS_DEPLOYMENT_TARGET = [0-9.]*/IPHONEOS_DEPLOYMENT_TARGET = 15.0/g' {} +
          npx cap sync ios
      - name: Install CocoaPods
        script: |
          cd ios/App
          pod install
      - name: Keychain initialize
        script: keychain initialize
      - name: Signing Setup
        script: |
          # 1. Lager sertifikat og privat nøkkel i ett steg og lagrer i keychain (--save)
          app-store-connect certificates create --type IOS_DISTRIBUTION --save
          
          # 2. Lager/henter profil (Den vil nå finne sertifikatet vi nettopp lagde)
          app-store-connect fetch-signing-files "$BUNDLE_ID" --type IOS_APP_STORE --create
          
          # 3. Bekrefter at alt er i nøkkelringen for Xcode
          keychain add-certificates
          xcode-project use-profiles
      - name: Build iOS IPA
        script: |
          # Bygger IPA med automatisk kobling av profiler
          xcode-project build-ipa \
            --workspace "$XCODE_WORKSPACE" \
            --scheme "$XCODE_SCHEME"
    artifacts:
      - build/ios/ipa/*.ipa
    publishing:
      app_store_connect:
        auth: integration
        submit_to_testflight: true
