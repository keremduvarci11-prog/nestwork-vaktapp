KOPIER OG SEND NØYAKTIG DETTE TIL AGENTEN NÅ:
"Nå skal alt fikses i én vending. Erstatt HELE codemagic.yaml med denne koden. Denne inneholder rettelsene for både sertifikat-nøkler, manglende profiler og korrekte kommando-argumenter:

YAML
workflows:
  ios-release:
    name: iOS Release Build
    instance_type: mac_mini_m2
    integrations:
      app_store_connect: Codemagic
    environment:
      node: 20
      vars:
        XCODE_WORKSPACE: "ios/App/App.xcworkspace"
        XCODE_SCHEME: "App"
        BUNDLE_ID: "no.nestwork.vaktapp"
    scripts:
      - name: Install npm dependencies
        script: npm install
      - name: Build web assets
        script: npm run build
      - name: Sync Capacitor
        script: |
          rm -rf ios
          npx cap add ios
          find ios -type f -name "Podfile" -exec sed -i '' "s/platform :ios, '[0-9.]*'/platform :ios, '15.0'/g" {} +
          find ios -type f -name "*.pbxproj" -exec sed -i '' 's/IPHONEOS_DEPLOYMENT_TARGET = [0-9.]*/IPHONEOS_DEPLOYMENT_TARGET = 15.0/g' {} +
          npx cap sync ios
      - name: Install CocoaPods
        script: |
          cd ios/App
          pod install
      - name: Keychain initialize
        script: keychain initialize
      - name: Full Signing and Project Setup
        script: |
          # 1. Lager sertifikat og privat nøkkel i ett steg (Fikser 'Missing Private Key')
          app-store-connect certificates create --type IOS_DISTRIBUTION --save
          
          # 2. Henter eller lager profil (Fikser 'No matching profile')
          app-store-connect fetch-signing-files "$BUNDLE_ID" --type IOS_APP_STORE --create
          
          # 3. Legger alt til i nøkkelringen
          keychain add-certificates
          
          # 4. Kobler profilene direkte til Xcode-prosjektet (Fikser 'Error 65 / Provisioning profile error')
          xcode-project use-profiles
      - name: Build iOS IPA
        script: |
          # Bygger IPA uten ekstra ugyldige argumenter
          xcode-project build-ipa \
            --workspace "$XCODE_WORKSPACE" \
            --scheme "$XCODE_SCHEME"
    artifacts:
      - build/ios/ipa/*.ipa
    publishing:
      app_store_connect:
        auth: integration
        submit_to_testflight: true
