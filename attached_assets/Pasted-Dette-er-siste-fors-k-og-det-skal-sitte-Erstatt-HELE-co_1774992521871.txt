Dette er siste forsøk og det skal sitte. Erstatt HELE codemagic.yaml med denne koden. Vi bruker nå en metode som tvinger den private nøkkelen inn i systemet manuelt før vi henter filer fra Apple:

YAML
workflows:
  ios-release:
    name: iOS Release Build
    instance_type: mac_mini_m2
    integrations:
      app_store_connect: Codemagic
    environment:
      node: 20
      vars:
        XCODE_WORKSPACE: "ios/App/App.xcworkspace"
        XCODE_SCHEME: "App"
        BUNDLE_ID: "no.nestwork.vaktapp"
    scripts:
      - name: Install npm dependencies
        script: npm install
      - name: Build web assets
        script: npm run build
      - name: Sync Capacitor
        script: |
          rm -rf ios
          npx cap add ios
          find ios -type f -name "Podfile" -exec sed -i '' "s/platform :ios, '[0-9.]*'/platform :ios, '15.0'/g" {} +
          find ios -type f -name "*.pbxproj" -exec sed -i '' 's/IPHONEOS_DEPLOYMENT_TARGET = [0-9.]*/IPHONEOS_DEPLOYMENT_TARGET = 15.0/g' {} +
          npx cap sync ios
      - name: Install CocoaPods
        script: |
          cd ios/App
          pod install
      - name: Keychain initialize
        script: keychain initialize
      - name: Force Signing Setup (Manual Key Injection)
        script: |
          # 1. Generer privat nøkkel
          openssl genrsa -out ios_dist.key 2048
          
          # 2. Importer nøkkelen manuelt i nøkkelringen (Dette er det manglende leddet!)
          keychain add-certificates --key ios_dist.key
          
          # 3. Lag sertifikat og profil hos Apple (Nå vil den finne nøkkelen i nøkkelringen)
          app-store-connect fetch-signing-files "$BUNDLE_ID" --type IOS_APP_STORE --create --delete-stale-profiles
          
          # 4. Bekreft for Xcode
          keychain add-certificates
          xcode-project use-profiles
      - name: Build iOS IPA
        script: |
          xcode-project build-ipa \
            --workspace "$XCODE_WORKSPACE" \
            --scheme "$XCODE_SCHEME"
    artifacts:
      - build/ios/ipa/*.ipa
    publishing:
      app_store_connect:
        auth: integration
        submit_to_testflight: true
