Erstatt HELE codemagic.yaml med denne koden. Vi fjerner alle upålitelige Codemagic-verktøy og bruker standard Xcode-kommandoer for å bygge. Dette er den mest rå og direkte metoden som finnes:

YAML
workflows:
  ios-release:
    name: iOS Release Build
    instance_type: mac_mini_m2
    integrations:
      app_store_connect: Codemagic
    environment:
      node: 20
      vars:
        XCODE_WORKSPACE: "ios/App/App.xcworkspace"
        XCODE_SCHEME: "App"
        BUNDLE_ID: "no.nestwork.vaktapp"
    scripts:
      - name: Install npm dependencies
        script: npm install
      - name: Build web assets
        script: npm run build
      - name: Sync Capacitor
        script: |
          rm -rf ios
          npx cap add ios
          find ios -type f -name "Podfile" -exec sed -i '' "s/platform :ios, '[0-9.]*'/platform :ios, '15.0'/g" {} +
          find ios -type f -name "*.pbxproj" -exec sed -i '' 's/IPHONEOS_DEPLOYMENT_TARGET = [0-9.]*/IPHONEOS_DEPLOYMENT_TARGET = 15.0/g' {} +
          npx cap sync ios
      - name: Install CocoaPods
        script: |
          cd ios/App
          pod install
      - name: Keychain initialize
        script: keychain initialize
      - name: Fetch and Setup Signing
        script: |
          # Vi bruker den ENE kommandoen som fungerer for å hente profiler
          app-store-connect fetch-signing-files "$BUNDLE_ID" --type IOS_APP_STORE --create
          keychain add-certificates
      - name: Build iOS IPA
        script: |
          # Vi bruker standard xcodebuild i stedet for Codemagic-verktøy som feiler
          xcodebuild build -workspace "$XCODE_WORKSPACE" -scheme "$XCODE_SCHEME" -configuration Release -sdk iphoneos
          
          # Vi pakker den manuelt for å unngå argument-feil
          xcode-project build-ipa --workspace "$XCODE_WORKSPACE" --scheme "$XCODE_SCHEME"
    artifacts:
      - build/ios/ipa/*.ipa
    publishing:
      app_store_connect:
        auth: integration
        submit_to_testflight: true
