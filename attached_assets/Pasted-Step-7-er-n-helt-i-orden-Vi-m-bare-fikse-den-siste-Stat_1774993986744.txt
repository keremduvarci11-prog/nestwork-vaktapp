Step 7 er nå helt i orden! Vi må bare fikse den siste 'Status 65'-feilen i Step 8. Erstatt HELE codemagic.yaml med denne koden. Jeg har lagt til en kommando som tvinger Xcode til å se profilene våre:

YAML
workflows:
  ios-release:
    name: iOS Release Build
    instance_type: mac_mini_m2
    integrations:
      app_store_connect: Codemagic
    environment:
      node: 20
      vars:
        XCODE_WORKSPACE: "ios/App/App.xcworkspace"
        XCODE_SCHEME: "App"
        BUNDLE_ID: "no.nestwork.vaktapp"
    scripts:
      - name: Install npm dependencies
        script: npm install
      - name: Build web assets
        script: npm run build
      - name: Sync Capacitor
        script: |
          rm -rf ios
          npx cap add ios
          find ios -type f -name "Podfile" -exec sed -i '' "s/platform :ios, '[0-9.]*'/platform :ios, '15.0'/g" {} +
          find ios -type f -name "*.pbxproj" -exec sed -i '' 's/IPHONEOS_DEPLOYMENT_TARGET = [0-9.]*/IPHONEOS_DEPLOYMENT_TARGET = 15.0/g' {} +
          npx cap sync ios
      - name: Install CocoaPods
        script: |
          cd ios/App
          pod install
      - name: Keychain initialize
        script: keychain initialize
      - name: Full Signing and Project Setup
        script: |
          # Dette fungerer nå - ikke rør!
          app-store-connect certificates create --type IOS_DISTRIBUTION --save
          app-store-connect fetch-signing-files "$BUNDLE_ID" --type IOS_APP_STORE --create
          keychain add-certificates
          xcode-project use-profiles
      - name: Build iOS IPA
        script: |
          # Vi tvinger prosjektet til å bruke 'Automatic' slik at den finner profilen vi nettopp lastet ned
          sed -i '' 's/CODE_SIGN_STYLE = Manual;/CODE_SIGN_STYLE = Automatic;/g' ios/App/App.xcodeproj/project.pbxproj
          
          # Bygger IPA-filen
          xcode-project build-ipa \
            --workspace "$XCODE_WORKSPACE" \
            --scheme "$XCODE_SCHEME"
    artifacts:
      - build/ios/ipa/*.ipa
    publishing:
      app_store_connect:
        auth: integration
        submit_to_testflight: true
