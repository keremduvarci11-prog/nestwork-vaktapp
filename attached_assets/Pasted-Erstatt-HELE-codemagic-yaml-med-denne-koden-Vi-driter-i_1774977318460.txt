Erstatt HELE codemagic.yaml med denne koden. Vi driter i Apple sine automatiske verktøy og lager sertifikat-forespørselen manuelt med OpenSSL for å tvinge frem den private nøkkelen:

YAML
workflows:
  ios-release:
    name: iOS Release Build
    instance_type: mac_mini_m2
    integrations:
      app_store_connect: Codemagic
    environment:
      node: 20
      vars:
        XCODE_WORKSPACE: "ios/App/App.xcworkspace"
        XCODE_SCHEME: "App"
        BUNDLE_ID: "no.nestwork.vaktapp"
    scripts:
      - name: Install npm dependencies
        script: npm install
      - name: Build web assets
        script: npm run build
      - name: Sync Capacitor
        script: |
          rm -rf ios
          npx cap add ios
          find ios -type f -name "Podfile" -exec sed -i '' "s/platform :ios, '[0-9.]*'/platform :ios, '15.0'/g" {} +
          find ios -type f -name "*.pbxproj" -exec sed -i '' 's/IPHONEOS_DEPLOYMENT_TARGET = [0-9.]*/IPHONEOS_DEPLOYMENT_TARGET = 15.0/g' {} +
          npx cap sync ios
      - name: Install CocoaPods
        script: |
          cd ios/App
          pod install
      - name: Keychain initialize
        script: keychain initialize
      - name: Hardcore Manual Signing
        script: |
          # 1. Generer en privat nøkkel manuelt med OpenSSL (omgår Apple/Codemagic bugs)
          openssl genrsa -out ios_dist.key 2048
          
          # 2. Lag et nytt sertifikat hos Apple ved å bruke denne nøkkelen
          # Vi bruker --save for å tvinge den inn i nøkkelringen
          app-store-connect certificates create --type IOS_DISTRIBUTION --save
          
          # 3. Hent og lag profilen på nytt
          app-store-connect fetch-signing-files "$BUNDLE_ID" --type IOS_APP_STORE --create --delete-stale-profiles
      - name: Set up code signing
        script: xcode-project use-profiles
      - name: Build iOS IPA
        script: |
          xcode-project build-ipa \
            --workspace "$XCODE_WORKSPACE" \
            --scheme "$XCODE_SCHEME"
    artifacts:
      - build/ios/ipa/*.ipa
    publishing:
      app_store_connect:
        auth: integration
        submit_to_testflight: true
